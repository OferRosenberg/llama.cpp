set(QAIRT_SDK_ROOT "" CACHE PATH "QAIRT SDK root containing include/QNN")
if (NOT EXISTS "${QAIRT_SDK_ROOT}/include/QNN/QnnInterface.h")
    message(FATAL_ERROR "GGML_QAIRT requires QAIRT_SDK_ROOT with QNN headers")
endif()
if (NOT WIN32)
    message(FATAL_ERROR "QAIRT M0 currently implements the Windows loader only")
endif()
if (GGML_QAIRT_HW_TESTS AND NOT GGML_QAIRT_TESTS)
    message(FATAL_ERROR "GGML_QAIRT_HW_TESTS requires GGML_QAIRT_TESTS")
endif()
if (GGML_QAIRT_TESTS AND NOT LLAMA_BUILD_TESTS)
    message(FATAL_ERROR "GGML_QAIRT_TESTS requires LLAMA_BUILD_TESTS")
endif()
if (GGML_QAIRT_TESTS AND DEFINED LLAMA_BUILD_COMMON AND NOT LLAMA_BUILD_COMMON)
    message(FATAL_ERROR "GGML_QAIRT_TESTS requires LLAMA_BUILD_COMMON")
endif()

add_library(ggml-qairt-core STATIC qairt-target.cpp qairt-runtime.cpp qairt-tensor.cpp qairt-graph.cpp qairt-lowering.cpp qairt-weights.cpp)
target_compile_features(ggml-qairt-core PRIVATE cxx_std_17)
target_include_directories(ggml-qairt-core PRIVATE "${QAIRT_SDK_ROOT}/include/QNN" ..)
target_link_libraries(ggml-qairt-core PRIVATE ggml-base Threads::Threads)
set_target_properties(ggml-qairt-core PROPERTIES POSITION_INDEPENDENT_CODE ON)

ggml_add_backend_library(ggml-qairt ggml-qairt.cpp qairt-backend.cpp qairt-partition.cpp)
target_compile_features(ggml-qairt PRIVATE cxx_std_17)
target_include_directories(ggml-qairt PRIVATE "${QAIRT_SDK_ROOT}/include/QNN")
if (NOT TARGET ggml-cpu)
    message(FATAL_ERROR "GGML_QAIRT requires GGML_CPU for singleton partition execution")
endif()
get_target_property(QAIRT_CPU_LIBRARY_TYPE ggml-cpu TYPE)
if (QAIRT_CPU_LIBRARY_TYPE STREQUAL "MODULE_LIBRARY")
    add_dependencies(ggml-qairt ggml-cpu)
    target_link_libraries(ggml-qairt PRIVATE "$<TARGET_FILE_DIR:ggml-cpu>/ggml-cpu.lib")
else()
    target_link_libraries(ggml-qairt PRIVATE ggml-cpu)
endif()
target_link_libraries(ggml-qairt PRIVATE ggml-qairt-core Threads::Threads)
